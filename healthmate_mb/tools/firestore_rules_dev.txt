// Development-friendly Firestore rules example
// WARNING: These rules are permissive for reads and should NOT be used in production
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Allow any client to read in_app_messages so the app can fetch campaigns
    match /in_app_messages/{docId} {
      allow read: if true;
      // Allow write only to authenticated users with admin claim (use server SDK to set admin=True)
      allow write: if request.auth != null && request.auth.token.admin == true;
    }

    // Users collection: allow clients to create their own document (optional)
    match /users/{userId} {
      allow create: if true; // allow unauthenticated client writes (we store simple profile docs)
      allow read, update, delete: if false; // disallow reads/updates from client apps for privacy in dev
    }
  }
}
